-- escaping quotes stops the string attack --
   SELECT * FROM users WHERE id='admin' AND pwd='x'' OR ''1''=''1'
   login accepted: false

   SELECT COUNT(*) FROM bus_pass WHERE route_no=99 OR 1=1
   rows returned for route 99 (should be 0): 3
   PreparedStatement, same input as a value: 0 rows
